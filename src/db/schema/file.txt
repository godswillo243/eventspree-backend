export const bookingStatusEnum = pgEnum("booking_status", [
  "pending_payment",
  "confirmed",
  "completed",
  "disputed",
  "refunded",
  "cancelled",
]);

export const bookings = pgTable(
  "bookings",
  {
    id: uuid("id").defaultRandom().primaryKey(),
    inquiryId: uuid("inquiry_id")
      .notNull()
      .unique()
      .references(() => inquiries.id),
    eventId: uuid("event_id")
      .notNull()
      .references(() => events.id),
    organizerId: uuid("organizer_id")
      .notNull()
      .references(() => users.id),

    targetType: targetTypeEnum("target_type").notNull(),
    vendorListingId: uuid("vendor_listing_id").references(
      () => vendorListings.id
    ),
    venueId: uuid("venue_id").references(() => venues.id),

    // Denormalized — resolved at booking-creation time from either
    // vendorListings.vendorProfileId → vendorProfiles.userId,
    // or venues.venueProviderId → venueProviderProfiles.userId.
    providerId: uuid("provider_id")
      .notNull()
      .references(() => users.id),

    agreedPrice: integer("agreed_price").notNull(),
    commissionRate: numeric("commission_rate", { precision: 5, scale: 4 })
      .notNull()
      .default("0.05"),
    commissionAmount: integer("commission_amount").notNull(),
    payoutAmount: integer("payout_amount").notNull(),

    status: bookingStatusEnum("status").notNull().default("pending_payment"),
    vendorMarkedCompleteAt: timestamp("vendor_marked_complete_at", {
      withTimezone: true,
    }),
    organizerConfirmedCompleteAt: timestamp("organizer_confirmed_complete_at", {
      withTimezone: true,
    }),

    confirmedAt: timestamp("confirmed_at", { withTimezone: true }),
    cancelledAt: timestamp("cancelled_at", { withTimezone: true }),
    createdAt: timestamp("created_at", { withTimezone: true })
      .notNull()
      .defaultNow(),
  },
  (t) => [
    index("bookings_vendor_listing_idx").on(t.vendorListingId),
    index("bookings_venue_idx").on(t.venueId),
    index("bookings_organizer_idx").on(t.organizerId),
    index("bookings_provider_idx").on(t.providerId),
    index("bookings_status_idx").on(t.status),
    check(
      "bookings_exactly_one_target",
      sql`(vendor_listing_id IS NOT NULL AND venue_id IS NULL) OR (vendor_listing_id IS NULL AND venue_id IS NOT NULL)`
    ),
    check(
      "bookings_target_matches_type",
      sql`(target_type = 'vendor' AND vendor_listing_id IS NOT NULL AND venue_id IS NULL)
   OR (target_type = 'venue' AND venue_id IS NOT NULL AND vendor_listing_id IS NULL)`
    ),
  ]
);

  (t) => [
    uniqueIndex("payments_reference_idx").on(t.reference),
    index("payments_booking_idx").on(t.bookingId),
    check(
      "payments_purpose_reference_check",
      sql`(
        ${t.purpose} = 'booking'
        AND ${t.bookingId} IS NOT NULL
      ) OR (
        ${t.purpose} = 'featured_listing'
        AND ${t.bookingId} IS NULL
      )`
    ),
  ]

  organizerProfile: one(organizerProfiles, {
    fields: [users.id],
    references: [organizerProfiles.userId],
  }),
  vendorProfile: one(vendorProfiles, {
    fields: [users.id],
    references: [vendorProfiles.userId],
  }),
  venueProviderProfile: one(venueProviderProfiles, {
    fields: [users.id],
    references: [venueProviderProfiles.userId],
  }),

  eventsAsOrganizer: many(events),
  inquiriesAsOrganizer: many(inquiries),
  messagesSent: many(messages),

  bookingsAsProvider: many(bookings, { relationName: "bookingProvider" }),

// ─────────────────────────────────────────────
export const inquiriesRelations = relations(inquiries, ({ one, many }) => ({
  event: one(events, {
    fields: [inquiries.eventId],
    references: [events.id],
  }),
  organizer: one(users, {
    fields: [inquiries.organizerId],
    references: [users.id],
  }),
  vendorListing: one(vendorListings, {
    fields: [inquiries.vendorListingId],
    references: [vendorListings.id],
  }),
  venue: one(venues, {
    fields: [inquiries.venueId],
    references: [venues.id],
  }),
  messages: many(messages),
  booking: one(bookings, {
    fields: [inquiries.id],
    references: [bookings.inquiryId],
  }),
}));

// ─────────────────────────────────────────────
// MESSAGES
// ─────────────────────────────────────────────
export const messagesRelations = relations(messages, ({ one }) => ({
  inquiry: one(inquiries, {
    fields: [messages.inquiryId],
    references: [inquiries.id],
  }),
  sender: one(users, {
    fields: [messages.senderId],
    references: [users.id],
  }),
}));

// ─────────────────────────────────────────────
// BOOKINGS (dual-target + dual-user relation)
// ─────────────────────────────────────────────
export const bookingsRelations = relations(bookings, ({ one, many }) => ({
  inquiry: one(inquiries, {
    fields: [bookings.inquiryId],
    references: [inquiries.id],
  }),
  event: one(events, {
    fields: [bookings.eventId],
    references: [events.id],
  }),
  organizer: one(users, {
    fields: [bookings.organizerId],
    references: [users.id],
    relationName: "bookingOrganizer",
  }),
  provider: one(users, {
    fields: [bookings.providerId],
    references: [users.id],
    relationName: "bookingProvider",
  }),
  vendorListing: one(vendorListings, {
    fields: [bookings.vendorListingId],
    references: [vendorListings.id],
  }),
  venue: one(venues, {
    fields: [bookings.venueId],
    references: [venues.id],
  }),
  payments: many(payments),
  review: many(reviews),
}));

import {
  check,
  index,
  integer,
  pgTable,
  timestamp,
  uniqueIndex,
  uuid,
  varchar,

import { bookings } from "./bookings";

import { targetTypeEnum } from "./enums";

export const reviews = pgTable(
  "reviews",
  {
    id: uuid("id").defaultRandom().primaryKey(),
    bookingId: uuid("booking_id")
      .notNull()
      .references(() => bookings.id, { onDelete: "cascade" }),
    reviewerId: uuid("reviewer_id")
      .notNull()
      .references(() => users.id),

    targetType: targetTypeEnum("target_type").notNull(),
    vendorListingId: uuid("vendor_listing_id").references(
      () => vendorListings.id
    ),
    venueId: uuid("venue_id").references(() => venues.id),

    rating: integer("rating").notNull(),
    comment: varchar("comment", { length: 2000 }),
    createdAt: timestamp("created_at", { withTimezone: true })
      .notNull()
      .defaultNow(),
  },
  (t) => [
    uniqueIndex("reviews_booking_reviewer_unique").on(
      t.bookingId,
      t.reviewerId
    ),
    index("reviews_vendor_listing_idx").on(t.vendorListingId),
    index("reviews_venue_idx").on(t.venueId),
    check("reviews_rating_range", sql`rating BETWEEN 1 AND 5`),
    check(
      "reviews_exactly_one_target",
      sql`(vendor_listing_id IS NOT NULL AND venue_id IS NULL) OR (vendor_listing_id IS NULL AND venue_id IS NOT NULL)`
    ),
    check(
      "reviews_target_matches_type",
      sql`(target_type = 'vendor' AND vendor_listing_id IS NOT NULL AND venue_id IS NULL)
   OR (target_type = 'venue' AND venue_id IS NOT NULL AND vendor_listing_id IS NULL)`
    ),
  ]
);

  boolean,
  check,
  index,

} from "drizzle-orm/pg-core";

export const priceUnitEnum = pgEnum("price_unit_enum", [
  "fixed",
  "per_day",
  "per_hour",
  "per_guest",
]);

export const vendorListings = pgTable(
  "vendor_listings",
  {
    id: uuid("id").defaultRandom().primaryKey(),
    vendorProfileId: uuid("vendor_profile_id")
      .notNull()
      .references(() => vendorProfiles.id, { onDelete: "cascade" }),
    categoryId: uuid("category_id")
      .notNull()
      .references(() => vendorListingsCategories.id, { onDelete: "restrict" }),

    title: varchar("title", { length: 255 }).notNull(), // "Premium Wedding Photography Package"
    description: varchar("description", { length: 2000 }),
    priceMin: integer("price_min"),
    priceMax: integer("price_max"),
    priceUnit: priceUnitEnum("price_unit").notNull().default("fixed"), // fixed | per_hour | per_guest | per_day

    portfolioImages: jsonb("portfolio_images")
      .$type<string[]>()
      .notNull()
      .default([]),
    isActive: boolean("is_active").notNull().default(true),

    ratingAvg: numeric("rating_avg", { precision: 3, scale: 2 }).default("0"),
    ratingCount: integer("rating_count").notNull().default(0),

    createdAt: timestamp("created_at", { withTimezone: true })
      .notNull()
      .defaultNow(),
    updatedAt: timestamp("updated_at", { withTimezone: true })
      .notNull()
      .defaultNow(),
  },
  (t) => [
    index("vendor_listings_vendor_idx").on(t.vendorProfileId),
    index("vendor_listings_category_idx").on(t.categoryId),
    index("vendor_listings_price_min_idx").on(t.priceMin),
    index("vendor_listings_price_max_idx").on(t.priceMax),
    index("vendor_listings_rating_avg_idx").on(t.ratingAvg),
    check(
      "vendor_listings_price_range_check",
      sql`(
    price_min IS NULL
    OR price_min >= 0
  ) AND (
    price_max IS NULL
    OR price_max >= 0
  ) AND (
    price_min IS NULL
    OR price_max IS NULL
    OR price_min <= price_max
  )`,
    ),
    check(
      "rating_avg_check",
      sql`
      (rating_avg >= 0) AND (rating_avg <= 5)
      `,
    ),
    check(
      "rating_count_check",
      sql`
      rating_count >= 0
      `,
    ),
  ],
);

// venues.ts
export const venues = pgTable(
  "venues",
  {
    id: uuid("id").defaultRandom().primaryKey(),
    venueProviderId: uuid("venue_provider_id")
      .notNull()
      .references(() => venueProviderProfiles.id, { onDelete: "cascade" }),

    name: varchar("name", { length: 255 }).notNull(),
    description: varchar("description", { length: 2000 }),
    venueTypeId: uuid("venue_type_id")
      .notNull()
      .references(() => categories.id, { onDelete: "restrict" }), // single category, appliesTo='venue'

    address: varchar("address", { length: 500 }).notNull(),
    state: varchar("state", { length: 100 }),
    city: varchar("city", { length: 100 }),

    capacityMin: integer("capacity_min"),
    capacityMax: integer("capacity_max"),
    amenities: jsonb("amenities").$type<string[]>().notNull().default([]),

    priceMin: integer("price_min"),
    priceMax: integer("price_max"),

    portfolioImages: jsonb("portfolio_images")
      .$type<string[]>()
      .notNull()
      .default([]),
    blockedDates: jsonb("blocked_dates")
      .$type<string[]>()
      .notNull()
      .default([]),

    // Each physical venue is independently approved — a provider
    // with 3 venues could have 2 approved and 1 pending/rejected.
    verificationStatus: verificationStatusEnum("verification_status")
      .notNull()
      .default("pending"),
    rejectionReason: varchar("rejection_reason", { length: 1000 }),

    featuredUntil: timestamp("featured_until", { withTimezone: true }),
    ratingAvg: numeric("rating_avg", { precision: 3, scale: 2 }).default("0"),

    createdAt: timestamp("created_at", { withTimezone: true })
      .notNull()
      .defaultNow(),
  },
  (t) => [
    index("venues_provider_idx").on(t.venueProviderId),
    index("venues_status_idx").on(t.verificationStatus),
    index("venues_venue_type_idx").on(t.venueTypeId),
    check(
      "venues_capacity_range_check",
      sql`(
    capacity_min IS NULL
    OR capacity_min >= 0
  ) AND (
    capacity_max IS NULL
    OR capacity_max >= 0
  ) AND (
    capacity_min IS NULL
    OR capacity_max IS NULL
    OR capacity_min <= capacity_max
  )`,
    ),
    check(
      "venues_price_range_check",
      sql`(
    price_min IS NULL
    OR price_min >= 0
  ) AND (
    price_max IS NULL
    OR price_max >= 0
  ) AND (
    price_min IS NULL
    OR price_max IS NULL
    OR price_min <= price_max
  )`,
    ),
    check(
      "rating_avg_check",
      sql`
      (rating_avg >= 0) AND (rating_avg <= 5)
      `,
    ),
  ],
);
